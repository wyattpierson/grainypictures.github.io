# E-mail alias commands
group E-mail Management

command List All E-mail Aliases
sendback perl -pe '$_ = "" if substr($_, 0, 1) eq "#" || $_ eq $/ || m/autoreply/' ~/etc/aliases\n
waitdiscard 30 0 Command finished.

command Add a New E-mail Alias
input name word New Alias:
input address string E-mail Address:
sendback touch ~/etc/aliases\n
waitdiscard 10 2 Command finished.
#sendback echo $name:$address >> ~/etc/aliases\n
#wait 10 2 Command finished.
sendbackblock %%%
perl -pi.bak - ~/etc/aliases
if (!$found) {
    if (q{$name} =~ /\@/ || q{$name} eq "") {
        print STDERR "The alias must be non-blank, and can't contain '\@'.\\n";
        $found = 1;
    } elsif (q{$address} eq "") {
        print STDERR "The address must be non-blank.\\n";
        $found = 1;
    }
    ($first, $rest) = split(":");
    if ($first eq q{$name}) {
        $_ = q{$name: $address} . $/;
        print STDERR q{Replaced existing entry for $name.} . $/;
        $found = 1;
    } elsif (eof) {
        s/\\n//; print $_ . $/ if $_ ne "";
        $_ = q{$name: $address} . $/;
        print STDERR q{Added $name.} . $/;
    }
}
__END__
%%%
waitdiscard 12 0 Command finished.
sendback vnewaliases\n
waitdiscard 10 0 Command finished.

command Delete an E-mail Alias
input alias word Alias to Delete:
sendbackblock %%%
perl -pi.bak - ~/etc/aliases
($first, $rest) = split(":");
if ($first eq q{$alias} && !m/autoreply/) {
    $_ = "";
    print STDERR q{Deleted $alias.} . $/;
    $found = 1;
} elsif (eof && !$found) {
    print STDERR "Couldn't find " . q{$alias} . ".\n";
}
__END__
%%%
waitdiscard 12 0 Command finished.
sendback vnewaliases\n
waitdiscard 10 0 Command finished.

group Cron Files (Automation)

command List the Current Cron Commands
#sendback crontab -l\n
sendback if (-e ~/etc/cronfile && !(-z ~/etc/cronfile)) perl -pe 'printf("%3d: ", $.)' ~/etc/cronfile\n
waitdiscard 10 0 Command finished.
sendback if (!(-e ~/etc/cronfile) || -z ~/etc/cronfile) echo "There are no commands in the cron table."\n
waitdiscard 10 0 Command finished.

command Add a New Command and Mail the Output
input hour word Hour, 0-23 or *:
input minute word Minute, 0-59 or *:
input weekday word Day of week, 0-7, Sun-Sat, or *:
input monthday word Day of month, 1-31 or *:
input month word Month, 1-12 or *:
input cmd string Command to execute:
input subject string Subject for output message:
input recipient string Recipient for output message:
sendback touch ~/etc/cronfile\n
waitdiscard 10 2 Command finished.
sendback echo "$minute $hour $monthday $month $weekday $cmd | /usr/bin/mail -s '$subject' $recipient" >> ~/etc/cronfile\n
waitdiscard 10 2 Command finished.
result Added the entry to the end of the cron table.  Installing the table.\n
sendback crontab ~/etc/cronfile\n
waitdiscard 10 0 Command finished.

command Add a New Command and Discard the Output
input hour word Hour, 0-23 or *:
input minute word Minute, 0-59 or *:
input weekday word Day of week, 0-7, Sun-Sat, or *:
input monthday word Day of month, 1-31 or *:
input month word Month, 1-12 or *:
input cmd string Command to execute:
sendback touch ~/etc/cronfile\n
waitdiscard 10 2 Command finished.
sendback echo "$minute $hour $monthday $month $weekday $cmd >/dev/null" >> ~/etc/cronfile\n
waitdiscard 10 2 Command finished.
result Added the entry to the end of the cron table.  Installing the table.\n
sendback crontab ~/etc/cronfile\n
waitdiscard 10 0 Command finished.

command Delete a Cron Command
input line number 0 32767 Line number to delete:
sendbackblock %%%
perl -pi.bak - ~/etc/cronfile
if ($. == $line) {
    $_ = "";
    print STDERR "Line deleted.\\n";
    $found = 1;
} elsif (eof && !$found) {
    print STDERR "Line not found in cron table.\\n";
}
__END__
%%%
waitdiscard 12 0 Command finished.
result Installing the table.\n
sendback crontab ~/etc/cronfile\n
waitdiscard 10 0 Command finished.

command Add a New MIME Type
input name word MIME Type:
input extensions string File Extensions:
sendback touch ~/www/conf/mime.types\n
waitdiscard 10 2 Command finished.
sendbackblock %%%
perl -pi.bak - ~/www/conf/mime.types
if (!$found) {
    ($first, $rest) = split(/ +/, $_, 2);
    $spaces = 31 > length(q{$name})? 31 - length(q{$name}) : 1;
    if ($first eq q{$name}) {
        $found = 1;
        $_ = q{$name} . sprintf("%${spaces}s", ' ') . q{$extensions} . $/;
        print STDERR q{Replaced existing entry for $name.} . $/;
    } elsif (eof) {
        s/\\n//; print $_ . $/ if $_ ne "";
        $_ = q{$name} . sprintf("%${spaces}s", ' ') . q{$extensions} . $/;
        print STDERR 'Added $name.' . $/;
    }
}
__END__
